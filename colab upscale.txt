import os
import cv2
from glob import glob

os.makedirs("/content/upscaled", exist_ok=True)

files = []
for ext in ("*.jpg", "*.jpeg", "*.png", "*.webp", "*.avif"):
    files.extend(glob(f"/content/hotel/{ext}"))

print(f"Found {len(files)} images")

for f in files:
    img = cv2.imread(f, cv2.IMREAD_UNCHANGED)

    if img is None:
        print("Skipped:", f)
        continue

    output, _ = upsampler.enhance(img, outscale=4)

    # JPEG doesn't support alpha channel
    if len(output.shape) == 3 and output.shape[2] == 4:
        output = cv2.cvtColor(output, cv2.COLOR_BGRA2BGR)

    name = os.path.splitext(os.path.basename(f))[0]
    out = os.path.join("/content/upscaled", name + ".jpg")

    cv2.imwrite(
        out,
        output,
        [cv2.IMWRITE_JPEG_QUALITY, 95]
    )
